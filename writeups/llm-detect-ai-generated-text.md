## Problem Description

Detect whether student essays were written by humans or generated by large language models (LLMs) - a critical challenge in educational integrity.

### The Challenge

- Binary classification: human (0) vs LLM-generated (1)
- Highly limited training data (only ~1,300 essays)
- LLM-generated text is increasingly human-like
- Robust to adversarial attacks and prompt variations

### Why It Matters

- **Academic Integrity**: Prevent AI-assisted cheating
- **Content Authentication**: Verify authorship
- **Misinformation**: Detect synthetic content
- **Education**: Preserve learning value of writing

```python
import openai
from transformers import AutoTokenizer, AutoModelForCausalLM

def generate_synthetic_essays(prompts, num_per_prompt=100):
    """Generate LLM-written essays for training"""
    synthetic_data = []

    for prompt in prompts:
        for _ in range(num_per_prompt):
            # Vary generation parameters for diversity
            temperature = np.random.uniform(0.7, 1.0)
            top_p = np.random.uniform(0.8, 0.95)

            essay = generate_essay(
                prompt,
                temperature=temperature,
                top_p=top_p,
                max_tokens=500
            )

            synthetic_data.append({
                'prompt': prompt,
                'text': essay,
                'generated': 1,  # LLM-generated
                'model': 'gpt-3.5',  # Track source
                'params': {'temp': temperature, 'top_p': top_p}
            })

    return synthetic_data

def generate_essay(prompt, model="gpt-3.5-turbo", **kwargs):
    """Generate single essay using OpenAI API"""
    response = openai.ChatCompletion.create(
        model=model,
        messages=[
            {"role": "system", "content": "You are a student writing an essay."},
            {"role": "user", "content": f"Write an essay about: {prompt}"}
        ],
        **kwargs
    )
    return response.choices[0].message.content
```

### 3. Statistical Text Features

```python
import numpy as np
from collections import Counter
import re

def extract_text_features(text):
    """Extract statistical features for AI detection"""
    features = {}

    # Basic statistics
    words = text.split()
    sentences = re.split(r'[.!?]+', text)

    features['word_count'] = len(words)
    features['sentence_count'] = len(sentences)
    features['avg_word_length'] = np.mean([len(w) for w in words])
    features['avg_sentence_length'] = np.mean([len(s.split()) for s in sentences if s])

    # Vocabulary richness
    unique_words = set(words)
    features['vocab_richness'] = len(unique_words) / len(words)
    features['hapax_legomena'] = sum(1 for w, c in Counter(words).items() if c == 1) / len(words)

    # Punctuation patterns
    features['comma_rate'] = text.count(',') / len(words)
    features['period_rate'] = text.count('.') / len(words)
    features['exclaim_rate'] = text.count('!') / len(words)

    # Sentence variety
    sentence_lengths = [len(s.split()) for s in sentences if s]
    features['sentence_length_std'] = np.std(sentence_lengths) if sentence_lengths else 0

    # N-gram analysis
    bigrams = list(zip(words[:-1], words[1:]))
    features['bigram_diversity'] = len(set(bigrams)) / (len(bigrams) + 1)

    # Repetition detection (LLMs sometimes repeat)
    features['repeated_phrases'] = count_repeated_phrases(text)

    return features

def count_repeated_phrases(text, min_length=3):
    """Count repeated multi-word phrases"""
    words = text.lower().split()
    phrases = []
    for length in range(min_length, 6):
        for i in range(len(words) - length + 1):
            phrase = tuple(words[i:i+length])
            phrases.append(phrase)

    counts = Counter(phrases)
    return sum(1 for c in counts.values() if c > 1)
```

### 4. Perplexity-Based Features

```python
from transformers import GPT2LMHeadModel, GPT2TokenizerFast
import torch
import math

def calculate_perplexity(text, model, tokenizer, stride=512):
    """
    Calculate perplexity using a language model
    Lower perplexity for LLM-generated text (more predictable)
    """
    encodings = tokenizer(text, return_tensors='pt')
    max_length = model.config.n_positions
    seq_len = encodings.input_ids.size(1)

    nlls = []
    prev_end_loc = 0

    for begin_loc in range(0, seq_len, stride):
        end_loc = min(begin_loc + max_length, seq_len)
        target_len = end_loc - prev_end_loc

        input_ids = encodings.input_ids[:, begin_loc:end_loc]
        target_ids = input_ids.clone()
        target_ids[:, :-target_len] = -100

        with torch.no_grad():
            outputs = model(input_ids, labels=target_ids)
            neg_log_likelihood = outputs.loss * target_len

        nlls.append(neg_log_likelihood)
        prev_end_loc = end_loc

        if end_loc == seq_len:
            break

    ppl = torch.exp(torch.stack(nlls).sum() / end_loc)
    return ppl.item()

# Usage
model = GPT2LMHeadModel.from_pretrained('gpt2')
tokenizer = GPT2TokenizerFast.from_pretrained('gpt2')

def get_perplexity_features(text):
    return {
        'perplexity': calculate_perplexity(text, model, tokenizer),
        'log_perplexity': math.log(calculate_perplexity(text, model, tokenizer) + 1)
    }
```

```python
class AIDetectionEnsemble:
    def __init__(self, models):
        """
        models: dict of {name: (model, weight)}
        """
        self.models = models

    def predict_proba(self, texts):
        """Weighted average of model predictions"""
        predictions = []
        weights = []

        for name, (model, weight) in self.models.items():
            pred = model.predict_proba(texts)[:, 1]
            predictions.append(pred)
            weights.append(weight)

        predictions = np.array(predictions)
        weights = np.array(weights) / sum(weights)

        # Weighted average
        final_pred = np.average(predictions, axis=0, weights=weights)
```
